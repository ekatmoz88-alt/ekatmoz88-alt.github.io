Endpoint-описание (как API)

endpoints:
  - name: "initialize_session"
    description: "Set substrate and spectral range"
    input:
      substrate:       { type: string, enum: [CaF2, BaF2, Ge] }
      spectrum_points: { type: int, range: [50, 250] }
      wavelength:      { type: array, range: [2.0, 20.0] }
      angle:           { type: float, default: 45.0 }
      polarization:    { type: enum, values: [S, P] }

  - name: "optimize"
    description: "Run numerical optimization"
    method: "random search → Powell"
    constraints:
      merit_function: "deviation of R from 50% (lower is better)"
      success_criteria: "value ≤ 2 (ideal < 1)"
    output: [R/T/A %, phase, .fil, xlsx]
